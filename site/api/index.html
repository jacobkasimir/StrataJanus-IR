    def tokenize(text: str) -> list[str]:
    """
    Convert text into tokens using stratajanus' version-1 rules.

    Steps:
    1. Unicode-normalize using NFC.
    2. Lowercase.
    3. Extract non-overlapping matches of Python's Unicode-aware ``\\w+``.
       Letters, digits, and underscore are retained within matches; other
       characters separate them. Repeated tokens and source order are kept.

    Args:
        text: Unicode string to analyze.

    Returns:
        Tokens in their order of appearance. Punctuation-only or empty input
        returns an empty list.

    Raises:
        TypeError: If ``text`` is not a string.
    """
    if not isinstance(text, str):
        raise TypeError(
            f"tokenize() expected str, got {type(text).__name__}"
        )

    normalized = unicodedata.normalize(
        CONFIG.unicode_normalization,
        text,
    )

    if CONFIG.lowercase:
        normalized = normalized.lower()

    return TOKEN_REGEX.findall(normalized)

    def verify_tokenizer_config(path: Path) -> None:
    """Check that a saved tokenizer configuration matches :data:`CONFIG`.

    Args:
        path: UTF-8 JSON configuration file to read.

    Returns:
        None when the decoded JSON equals the active configuration.

    Raises:
        FileNotFoundError: If the path does not exist.
        json.JSONDecodeError: If the file is not valid JSON.
        ValueError: If the saved settings differ from the active settings.
    """
    with path.open("r", encoding="utf-8") as file:
        saved_config = json.load(file)

    current_config = asdict(CONFIG)

    if saved_config != current_config:
        raise ValueError(
            "Tokenizer configuration does not match the index.\n"
            f"Saved:   {saved_config}\n"
            f"Current: {current_config}"
        )

    @dataclass
class InvertedIndex:
    """Hold postings and per-document statistics for an in-memory collection.

    Args:
        postings: Mapping from token strings to lists of
            ``(internal_id, term_frequency)`` pairs. Lists produced by
            :func:`build_index` follow ascending internal-ID order.
        doc_lengths: Token count for each internal document ID.
        internal_to_external: External ``doc_id`` for each internal ID.

    Internal IDs are zero-based positions shared by ``doc_lengths`` and
    ``internal_to_external``. External IDs are the caller's original strings.
    The dataclass does not validate manually supplied structures; builders
    return aligned lists and valid posting IDs.
    """

    postings: dict[str, list[tuple[int, int]]]
    doc_lengths: list[int]
    internal_to_external: list[str]

    @property
    def document_count(self) -> int:
        """Number of documents, including documents with no tokens."""
        return len(self.internal_to_external)

    @property
    def average_document_length(self) -> float:
        """Mean token count, or ``0.0`` for an empty collection."""
        if self.document_count == 0:
            return 0.0

        return sum(self.doc_lengths) / self.document_count

    def document_frequency(self, term: str) -> int:
        """Return the number of postings for the exact term; unknown terms yield 0."""
        return len(self.postings.get(term, []))

    def build_index(documents: list[dict]) -> InvertedIndex:
    """Tokenize documents and produce an in-memory inverted index.

    Args:
        documents: Ordered iterable of mappings with a nonempty string
            ``doc_id`` and string ``title`` and ``text`` fields. Extra fields
            are ignored.

    Returns:
        An :class:`InvertedIndex` whose internal IDs follow input order,
        lengths count title and body tokens together, and postings contain
        ``(internal_id, term_frequency)`` pairs in that same order.

    Raises:
        TypeError: If an item is not a dictionary.
        ValueError: If required fields are invalid or an external ID repeats.

    Empty input produces an empty index. Input mappings are not modified.
    The title and body are separated by a newline before tokenization.
    """
    postings: dict[str, list[tuple[int, int]]] = defaultdict(list)
    doc_lengths: list[int] = []
    internal_to_external: list[str] = []
    seen_doc_ids: set[str] = set()

    for document in documents:
        if not isinstance(document, dict):
            raise TypeError("Each document must be a dictionary.")

        doc_id = document.get("doc_id")
        title = document.get("title")
        text = document.get("text")

        if not isinstance(doc_id, str) or not doc_id:
            raise ValueError("Document 'doc_id' must be a nonempty string.")

        if not isinstance(title, str):
            raise ValueError(
                f"Document {doc_id!r}: 'title' must be a string."
            )

        if not isinstance(text, str):
            raise ValueError(
                f"Document {doc_id!r}: 'text' must be a string."
            )

        if doc_id in seen_doc_ids:
            raise ValueError(f"Duplicate document ID: {doc_id!r}")

        internal_id = len(internal_to_external)
        seen_doc_ids.add(doc_id)

        # The separator prevents a word at the end of the title from
        # merging with a word at the beginning of the text.
        tokens = tokenize(title + "\n" + text)

        internal_to_external.append(doc_id)
        doc_lengths.append(len(tokens))

        term_frequencies = Counter(tokens)

        for term, frequency in term_frequencies.items():
            postings[term].append((internal_id, frequency))

    return InvertedIndex(
        postings=dict(postings),
        doc_lengths=doc_lengths,
        internal_to_external=internal_to_external,
    )

    class BM25:
    """Score matching documents with the Okapi BM25 formula.

    Args:
        index: In-memory index whose postings and document statistics are
            treated as read-only for this ranker's lifetime.
        k1: Positive finite term-frequency saturation parameter (default 1.2).
        b: Finite document-length normalization parameter in ``[0, 1]``
            (default 0.75).

    The implementation uses ``log(1 + (N - df + 0.5) / (df + 0.5))`` for
    IDF and sums the standard BM25 term contribution. Repeated normalized
    query terms count once. Results sort by descending score, with ties broken
    by ascending internal ID, preserving input order independently of external
    ID spelling. Zero-score documents are omitted.
    """

    def __init__(
        self,
        index: InvertedIndex,
        k1: float = 1.2,
        b: float = 0.75,
    ):
        """Create a ranker; invalid ``k1`` or ``b`` raises ``ValueError``."""
        if not math.isfinite(k1) or k1 <= 0:
            raise ValueError("k1 must be finite and greater than zero.")

        if not math.isfinite(b) or not 0 <= b <= 1:
            raise ValueError("b must be finite and between 0 and 1.")

        self.index = index
        self.k1 = k1
        self.b = b

        # Compute these once, rather than repeatedly during scoring.
        self.document_count = index.document_count
        self.average_document_length = index.average_document_length

    def search(
        self,
        query: str,
        top_k: int = 10,
    ) -> list[SearchResult]:
        """Return up to ``top_k`` positive-scoring matches for ``query``.

        Args:
            query: Text normalized with the index tokenizer.
            top_k: Maximum result count; nonpositive values return no results.

        Returns:
            Results containing internal ID, original external ``doc_id``, and
            BM25 score, in descending score order with deterministic ID ties.
            Empty/unknown queries and empty or all-zero-length collections
            return an empty list. Query terms are deduplicated after
            tokenization.

        Raises:
            TypeError: If ``query`` is not a string.
        """
        if top_k <= 0:
            return []

        # Sorting gives a deterministic accumulation order.
        query_terms = sorted(set(tokenize(query)))

        if not query_terms:
            return []

        if (
            self.document_count == 0
            or self.average_document_length == 0
        ):
            return []

        scores: dict[int, float] = defaultdict(float)

        for term in query_terms:
            postings = self.index.postings.get(term, [])

            if not postings:
                continue

            document_frequency = len(postings)

            idf = math.log1p(
                (
                    self.document_count
                    - document_frequency
                    + 0.5
                )
                / (document_frequency + 0.5)
            )

            for internal_id, term_frequency in postings:
                document_length = self.index.doc_lengths[internal_id]

                length_normalization = (
                    1
                    - self.b
                    + self.b
                    * document_length
                    / self.average_document_length
                )

                numerator = term_frequency * (self.k1 + 1)

                denominator = (
                    term_frequency
                    + self.k1 * length_normalization
                )

                scores[internal_id] += (
                    idf * numerator / denominator
                )

        # Smallest (-score, ID) means highest score, then lowest ID.
        best = heapq.nsmallest(
            top_k,
            scores.items(),
            key=lambda item: (-item[1], item[0]),
        )

        return [
            SearchResult(
                internal_id=internal_id,
                doc_id=self.index.internal_to_external[internal_id],
                score=score,
            )
            for internal_id, score in best
        ]

    """Load SciFact splits, validate their references, and print an example.

    Returns:
        None: The function prints split statistics and one judged example.

    Raises:
        FileNotFoundError: If no qrels splits exist or the train split is missing.
        AssertionError: If a loaded split is empty or contains dangling references.
        ImportError: If the optional ``beir`` dependency is unavailable.
    """
    split = "train"

    qrels_dir = DATASET_DIR / "qrels"
    available = sorted(path.stem for path in qrels_dir.glob("*.tsv"))
    if not available:
        raise FileNotFoundError(f"No qrels splits found in {qrels_dir}")
    print(f"Available relevance splits: {', '.join(available)}")

    if split not in available:
        raise FileNotFoundError(
            f"Split '{split}' is unavailable. Available splits: {available}"
        )

    corpus = queries = qrels = None
    for current_split in available:
        current_corpus, current_queries, current_qrels = GenericDataLoader(
            data_folder=str(DATASET_DIR)
        ).load(split=current_split)
        assert current_corpus, f"The corpus is empty for {current_split}."
        assert current_queries, f"No queries were loaded for {current_split}."
        assert current_qrels, f"No judgments were loaded for {current_split}."

        for query_id, judgments in current_qrels.items():
            assert query_id in current_queries, (
                f"{current_split} judgments reference missing query: {query_id}"
            )
            for doc_id in judgments:
                assert doc_id in current_corpus, (
                    f"{current_split} judgments reference missing document: {doc_id}"
                )

        print(
            f"{current_split}: {len(current_corpus):,} documents, "
            f"{len(current_queries):,} queries, "
            f"{len(current_qrels):,} queries with judgments"
        )
        if current_split == split:
            corpus, queries, qrels = current_corpus, current_queries, current_qrels

    print("\nDataset consistency checks passed for every available split.\n")

    print(f"Split: {split}")
    print(f"Documents: {len(corpus):,}")
    print(f"Queries loaded: {len(queries):,}")
    print(f"Queries with judgments: {len(qrels):,}")

    query_id = next(iter(qrels))

    print("\nEXAMPLE QUERY")
    print(f"ID: {query_id}")
    print(f"Text: {queries[query_id]}")

    print("\nRELEVANT DOCUMENTS")
    for doc_id, relevance in qrels[query_id].items():
        if relevance <= 0:
            continue

        document = corpus[doc_id]

        print(f"\nDocument ID: {doc_id}")

    def read_jsonl(path):
    """Read nonblank JSON Lines records with their source line numbers.

    Args:
        path (Path): UTF-8 JSONL file to read.

    Returns:
        Iterator[tuple[int, dict[str, Any]]]: Source line numbers and decoded objects.

    Raises:
        ValueError: If a nonblank line is invalid JSON or is not a JSON object.
    """
    with path.open("r", encoding="utf-8") as file:
        for line_number, line in enumerate(file, start=1):
            if not line.strip():
                continue

            try:
                row = json.loads(line)
            except json.JSONDecodeError as exc:
                raise ValueError(
                    f"Invalid JSON in {path}, line {line_number}"
                ) from exc

            if not isinstance(row, dict):
                raise ValueError(
                    f"Expected a JSON object in {path}, "
                    f"line {line_number}"
                )

            yield line_number, row

    def convert_documents():
    """Convert the raw SciFact corpus and create its external ID mapping.

    Returns:
        tuple[set[str], int]: Document IDs and the count of empty documents.

    Raises:
        ValueError: If a record is invalid or contains a duplicate document ID.
    """
    input_path = RAW_DIR / "corpus.jsonl"
    documents_path = OUTPUT_DIR / "documents.jsonl"
    mapping_path = OUTPUT_DIR / "doc_ids.jsonl"

    seen_ids = set()
    empty_documents = 0

    with (
        documents_path.open("w", encoding="utf-8") as documents_file,
        mapping_path.open("w", encoding="utf-8") as mapping_file,
    ):
        for line_number, row in read_jsonl(input_path):
            location = f"{input_path}:{line_number}"

            doc_id = require_id(row, "_id", location)
            title = require_text(row, "title", location, default="")
            text = require_text(row, "text", location)

            if doc_id in seen_ids:
                raise ValueError(
                    f"{location}: duplicate document ID '{doc_id}'"
                )

            internal_id = len(seen_ids)
            seen_ids.add(doc_id)

            if not title.strip() and not text.strip():
                empty_documents += 1

            document = {
                "doc_id": doc_id,
                "title": title,
                "text": text,
                "metadata": prepare_metadata(row, location),
            }

            mapping = {
                "internal_id": internal_id,
                "doc_id": doc_id,
            }

            write_jsonl_row(documents_file, document)
            write_jsonl_row(mapping_file, mapping)

    return seen_ids, empty_documents

    def convert_qrels(document_ids, query_ids):
    """Convert qrels splits and validate referenced queries and documents.

    Args:
        document_ids (set[str]): Valid document IDs from the converted corpus.
        query_ids (set[str]): Valid query IDs from the converted queries.

    Returns:
        dict[str, dict[str, int]]: Per-split counts of judgments, queries, and positives.

    Raises:
        ValueError: If qrels are missing, malformed, duplicated, or reference unknown IDs.
    """
    input_dir = RAW_DIR / "qrels"
    output_dir = OUTPUT_DIR / "qrels"

    output_dir.mkdir(parents=True, exist_ok=True)

    split_paths = sorted(input_dir.glob("*.tsv"))

    if not split_paths:
        raise ValueError(f"No qrels files found in {input_dir}")

    split_statistics = {}

    for input_path in split_paths:
        split_name = input_path.stem
        output_path = output_dir / input_path.name

        seen_pairs = set()
        split_query_ids = set()
        positive_count = 0

        with (
            input_path.open(
                "r", encoding="utf-8", newline=""
            ) as input_file,
            output_path.open(
                "w", encoding="utf-8", newline=""
            ) as output_file,
        ):
            reader = csv.DictReader(input_file, delimiter="\t")

            expected_fields = {"query-id", "corpus-id", "score"}
            actual_fields = set(reader.fieldnames or [])

            if not expected_fields.issubset(actual_fields):
                raise ValueError(
                    f"{input_path}: unexpected columns "
                    f"{reader.fieldnames}"
                )

            writer = csv.DictWriter(
                output_file,
                fieldnames=["query_id", "doc_id", "relevance"],
                delimiter="\t",
                lineterminator="\n",
            )
            writer.writeheader()

            for row in reader:
                location = f"{input_path}:{reader.line_num}"

                query_id = row["query-id"]
                doc_id = row["corpus-id"]

                try:
                    relevance = int(row["score"])
                except (ValueError, TypeError) as exc:
                    raise ValueError(
                        f"{location}: invalid relevance score"
                    ) from exc

                if query_id not in query_ids:
                    raise ValueError(
                        f"{location}: unknown query '{query_id}'"
                    )

                if doc_id not in document_ids:
                    raise ValueError(
                        f"{location}: unknown document '{doc_id}'"
                    )

                pair = (query_id, doc_id)

                if pair in seen_pairs:
                    raise ValueError(
                        f"{location}: duplicate judgment {pair}"
                    )

                seen_pairs.add(pair)
                split_query_ids.add(query_id)

                if relevance > 0:
                    positive_count += 1

                writer.writerow({
                    "query_id": query_id,
                    "doc_id": doc_id,
                    "relevance": relevance,
                })

        split_statistics[split_name] = {
            "judgments": len(seen_pairs),
            "queries": len(split_query_ids),
            "positive_judgments": positive_count,
        }

    return split_statistics

    # build_index accepts an iterable, so it can consume the
    # documents produced by read_jsonl one at a time.
    index = build_index(read_jsonl(documents_path))

    with manifest_path.open("r", encoding="utf-8") as file:
        manifest = json.load(file)

    expected_count = manifest["documents"]

    if index.document_count != expected_count:
        raise ValueError(
            f"Document count mismatch: index has "
            f"{index.document_count}, manifest says {expected_count}"
        )

    # Step 4 assigned internal IDs in corpus-file order.
    # Confirm that build_index assigned exactly the same IDs.
    mapping_count = 0

    for mapping in read_jsonl(mapping_path):
        expected_internal_id = mapping_count
        actual_internal_id = mapping["internal_id"]

        if actual_internal_id != expected_internal_id:
            raise ValueError(
                f"Mapping entry {mapping_count} has internal ID "
                f"{actual_internal_id}, expected {expected_internal_id}"
            )

        if expected_internal_id >= index.document_count:
            raise ValueError(
                "ID mapping contains more documents than the index"
            )

        expected_doc_id = mapping["doc_id"]
        actual_doc_id = index.internal_to_external[
            expected_internal_id
        ]

        if actual_doc_id != expected_doc_id:
            raise ValueError(
                f"ID mismatch at position {expected_internal_id}: "
                f"index has {actual_doc_id!r}, "
                f"mapping has {expected_doc_id!r}"
            )

        mapping_count += 1

    if mapping_count != index.document_count:
        raise ValueError(
            f"ID mapping has {mapping_count} documents; "
            f"index has {index.document_count}"
        )

    print("SciFact index checks passed.")
    print(f"Documents: {index.document_count:,}")
    print(f"Distinct terms: {len(index.postings):,}")
    print(
        "Average document length: "
        f"{index.average_document_length:.2f} tokens"
    )

    print("\nExample terms:")
    for term in ("science", "evidence", "protein"):
        postings = index.postings.get(term, [])
        print(
            f"  {term!r}: {len(postings)} documents; "
            f"first five postings: {postings[:5]}"
        )

    def read_documents(path: Path):
    """Yield nonblank JSONL document values in source order.

    Args:
        path: UTF-8 JSON Lines file containing processed documents.

    Yields:
        Decoded JSON values for the index builder.

    Raises:
        FileNotFoundError: If the file is missing.
        ValueError: If a nonblank line contains malformed JSON.
    """
    with path.open("r", encoding="utf-8") as file:
        for line_number, line in enumerate(file, start=1):
            if not line.strip():
                continue

            try:
                document = json.loads(line)
            except json.JSONDecodeError as exc:
                raise ValueError(
                    f"Invalid JSON in {path}, line {line_number}"
                ) from exc

            yield document

    def main():
    """Parse a query, build the SciFact index, and print ranked results.

    The command accepts one query argument and an optional ``--top-k`` count.
    It rebuilds the full in-memory index for each invocation and uses result
    internal IDs to retrieve the corresponding source documents for display.

    Raises:
        SystemExit: For command-line parsing errors.
        FileNotFoundError: If the processed SciFact corpus is not present.

    Side effects:
        Writes progress and ranked result details to standard output.
    """
    parser = argparse.ArgumentParser(
        description="Search SciFact using stratajanus BM25."
    )

    parser.add_argument(
        "query",
        help="Search query. Put multiword queries in quotes.",
    )

    parser.add_argument(
        "--top-k",
        type=int,
        default=10,
        help="Maximum number of results.",
    )

    args = parser.parse_args()

    if not DOCUMENTS_PATH.is_file():
        raise FileNotFoundError(
            f"Missing processed corpus: {DOCUMENTS_PATH}"
        )

    # SciFact is small enough to retain the documents for display.
    documents = list(read_documents(DOCUMENTS_PATH))

    print(f"Building index for {len(documents):,} documents...")

    index = build_index(documents)
    ranker = BM25(index)

    results = ranker.search(
        args.query,
        top_k=args.top_k,
    )

    print(f"\nQuery: {args.query}")
    print(f"Results returned: {len(results)}\n")

    for rank, result in enumerate(results, start=1):
        document = documents[result.internal_id]

        print(
            f"{rank}. {result.doc_id} "
            f"| BM25 score: {result.score:.4f}"
        )
        print(f"   {document['title']}")
        print()
