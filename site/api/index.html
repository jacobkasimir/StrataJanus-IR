    def tokenize(text: str) -> list[str]:
    """
    Convert text into tokens using stratajanus' version-1 rules.

    Steps:
    1. Unicode-normalize using NFC.
    2. Lowercase.
    3. Extract non-overlapping matches of Python's Unicode-aware ``\\w+``.
       Letters, digits, and underscore are retained within matches; other
       characters separate them. Repeated tokens and source order are kept.

    Args:
        text: Unicode string to analyze.

    Returns:
        Tokens in their order of appearance. Punctuation-only or empty input
        returns an empty list.

    Raises:
        TypeError: If ``text`` is not a string.
    """
    if not isinstance(text, str):
        raise TypeError(
            f"tokenize() expected str, got {type(text).__name__}"
        )

    normalized = unicodedata.normalize(
        CONFIG.unicode_normalization,
        text,
    )

    if CONFIG.lowercase:
        normalized = normalized.lower()

    return TOKEN_REGEX.findall(normalized)

    def verify_tokenizer_config(path: Path) -> None:
    """Check that a saved tokenizer configuration matches :data:`CONFIG`.

    Args:
        path: UTF-8 JSON configuration file to read.

    Returns:
        None when the decoded JSON equals the active configuration.

    Raises:
        FileNotFoundError: If the path does not exist.
        json.JSONDecodeError: If the file is not valid JSON.
        ValueError: If the saved settings differ from the active settings.

    with path.open("r", encoding="utf-8") as file:
        saved_config = json.load(file)

    current_config = asdict(CONFIG)

    if saved_config != current_config:
        raise ValueError(
            "Tokenizer configuration does not match the index.\n"
            f"Saved:   {saved_config}\n"
            f"Current: {current_config}"
        )

    @dataclass
class InvertedIndex:
    """Hold postings and per-document statistics for an in-memory collection.

    Args:
        postings: Mapping from token strings to lists of
            ``(internal_id, term_frequency)`` pairs. Lists produced by
            :func:`build_index` follow ascending internal-ID order.
        doc_lengths: Token count for each internal document ID.
        internal_to_external: External ``doc_id`` for each internal ID.

    Internal IDs are zero-based positions shared by ``doc_lengths`` and
    ``internal_to_external``. External IDs are the caller's original strings.
    The dataclass does not validate manually supplied structures; builders
    return aligned lists and valid posting IDs.
    """

    postings: dict[str, list[tuple[int, int]]]
    doc_lengths: list[int]
    internal_to_external: list[str]

    @property
    def document_count(self) -> int:
        """Number of documents, including documents with no tokens."""
        return len(self.internal_to_external)

    @property
    def average_document_length(self) -> float:
        """Mean token count, or ``0.0`` for an empty collection."""
        if self.document_count == 0:
            return 0.0

        return sum(self.doc_lengths) / self.document_count

    def document_frequency(self, term: str) -> int:
        """Return the number of postings for the exact term; unknown terms yield 0."""
        return len(self.postings.get(term, []))

    def build_index(documents: list[dict]) -> InvertedIndex:
    """Tokenize documents and produce an in-memory inverted index.

    Args:
        documents: Ordered iterable of mappings with a nonempty string
            ``doc_id`` and string ``title`` and ``text`` fields. Extra fields
            are ignored.

    Returns:
        An :class:`InvertedIndex` whose internal IDs follow input order,
        lengths count title and body tokens together, and postings contain
        ``(internal_id, term_frequency)`` pairs in that same order.

    Raises:
        TypeError: If an item is not a dictionary.
        ValueError: If required fields are invalid or an external ID repeats.

    Empty input produces an empty index. Input mappings are not modified.
    The title and body are separated by a newline before tokenization.
    """
    postings: dict[str, list[tuple[int, int]]] = defaultdict(list)
    doc_lengths: list[int] = []
    internal_to_external: list[str] = []
    seen_doc_ids: set[str] = set()

    for document in documents:
        if not isinstance(document, dict):
            raise TypeError("Each document must be a dictionary.")

        doc_id = document.get("doc_id")
        title = document.get("title")
        text = document.get("text")

        if not isinstance(doc_id, str) or not doc_id:
            raise ValueError("Document 'doc_id' must be a nonempty string.")

        if not isinstance(title, str):
            raise ValueError(
                f"Document {doc_id!r}: 'title' must be a string."
            )

        if not isinstance(text, str):
            raise ValueError(
                f"Document {doc_id!r}: 'text' must be a string."
            )

        if doc_id in seen_doc_ids:
            raise ValueError(f"Duplicate document ID: {doc_id!r}")

        internal_id = len(internal_to_external)
        seen_doc_ids.add(doc_id)

        # The separator prevents a word at the end of the title from
        # merging with a word at the beginning of the text.
        tokens = tokenize(title + "\n" + text)

        internal_to_external.append(doc_id)
        doc_lengths.append(len(tokens))

        term_frequencies = Counter(tokens)

        for term, frequency in term_frequencies.items():
            postings[term].append((internal_id, frequency))

    return InvertedIndex(
        postings=dict(postings),
        doc_lengths=doc_lengths,
        internal_to_external=internal_to_external,
    )

    def save_index(index: InvertedIndex, directory: Path) -> None:
    """
    Save a complete index to a new directory.

    Args:
        index: In-memory index to save.
        directory: Destination directory. It must not already exist.

    Raises:
        ValueError: If the index is structurally inconsistent.
        FileExistsError: If the destination already exists.

    The manifest is written last. A failed save may leave a partial
    directory that must be inspected before retrying.
    """
    directory = Path(directory)

    if directory.exists():
        raise FileExistsError(
            f"Index directory already exists: {directory}"
        )

    validate_index(index)

    directory.mkdir(parents=True, exist_ok=False)

    payload = {
        "postings": index.postings,
        "doc_lengths": index.doc_lengths,
        "internal_to_external": index.internal_to_external,
    }

    with (directory / "index.json").open(
        "w",
        encoding="utf-8",
    ) as file:
        json.dump(
            payload,
            file,
            ensure_ascii=False,
            sort_keys=True,
            indent=2,
        )
        file.write("\n")

    save_tokenizer_config(
        directory / "tokenizer_config.json"
    )

    manifest = {
        "format": FORMAT_NAME,
        "format_version": FORMAT_VERSION,
        "document_count": index.document_count,
        "term_count": len(index.postings),
    }

    # Write this only after the index and tokenizer settings exist.
    with (directory / "manifest.json").open(
        "w",
        encoding="utf-8",
    ) as file:
        json.dump(manifest, file, indent=2)
        file.write("\n")

    def load_index(directory: Path) -> InvertedIndex:
    """
    Load and validate a saved JSON index.

    Args:
        directory: Directory containing the saved index files.

    Returns:
        A reconstructed InvertedIndex.

    Raises:
        FileNotFoundError: If a required file is missing.
        ValueError: If the format, tokenizer configuration, or index
            structure is incompatible or invalid.
    """
    directory = Path(directory)

    required_files = [
        directory / "manifest.json",
        directory / "tokenizer_config.json",
        directory / "index.json",
    ]

    for path in required_files:
        if not path.is_file():
            raise FileNotFoundError(
                f"Required index file is missing: {path}"
            )

    with (directory / "manifest.json").open(
        "r",
        encoding="utf-8",
    ) as file:
        manifest = json.load(file)

    if not isinstance(manifest, dict):
        raise ValueError("Index manifest must be a JSON object.")

    if manifest.get("format") != FORMAT_NAME:
        raise ValueError("Unsupported index format.")

    if manifest.get("format_version") != FORMAT_VERSION:
        raise ValueError("Unsupported index format version.")

    verify_tokenizer_config(
        directory / "tokenizer_config.json"
    )

    with (directory / "index.json").open(
        "r",
        encoding="utf-8",
    ) as file:
        payload = json.load(file)

    try:
        index = InvertedIndex(
            postings={
                term: [tuple(posting) for posting in postings]
                for term, postings in payload["postings"].items()
            },
            doc_lengths=payload["doc_lengths"],
            internal_to_external=payload["internal_to_external"],
        )
    except (KeyError, TypeError, AttributeError) as exc:
        raise ValueError(
            "Saved index has an invalid structure."
        ) from exc

    validate_index(index)

    if index.document_count != manifest.get("document_count"):
        raise ValueError(
            "Index document count does not match the manifest."
        )

    if len(index.postings) != manifest.get("term_count"):
        raise ValueError(
            "Index term count does not match the manifest."
        )

    return index

    def save_index(index: InvertedIndex, directory: Path) -> None:
    """
    Save an index using fixed-width binary integers.

    Document IDs, term frequencies, and document lengths use
    unsigned 32-bit little-endian integers.

    Args:
        index: Complete in-memory inverted index.
        directory: New destination directory.

    Raises:
        FileExistsError: If the destination already exists.
        ValueError: If the index is invalid or contains values
            outside the supported integer range.

    A failed write may leave a partial directory. The manifest
    is written last.
    """
    directory = Path(directory)

    if directory.exists():
        raise FileExistsError(
            f"Index directory already exists: {directory}"
        )

    validate_index(index)

    for length in index.doc_lengths:
        _check_uint32(length, "Document length")

    for postings in index.postings.values():
        for internal_id, frequency in postings:
            _check_uint32(internal_id, "Internal document ID")
            _check_uint32(frequency, "Term frequency")

    directory.mkdir(parents=True, exist_ok=False)

    with (directory / "doc_lengths.bin").open("wb") as file:
        for length in index.doc_lengths:
            file.write(UINT32.pack(length))

    vocabulary = {}

    with (directory / "postings.bin").open("wb") as file:
        for term in sorted(index.postings):
            postings = index.postings[term]

            vocabulary[term] = {
                "offset": file.tell(),
                "document_frequency": len(postings),
            }

            for internal_id, frequency in postings:
                file.write(
                    POSTING.pack(internal_id, frequency)
                )

    _write_json(
        directory / "vocabulary.json",
        vocabulary,
    )

    _write_json(
        directory / "doc_ids.json",
        index.internal_to_external,
    )

    save_tokenizer_config(
        directory / "tokenizer_config.json"
    )

    _write_json(
        directory / "manifest.json",
        {
            "format": FORMAT_NAME,
            "format_version": FORMAT_VERSION,
            "document_count": index.document_count,
            "term_count": len(index.postings),
        },
    )

    def load_index(directory: Path) -> InvertedIndex:
    """
    Load and validate a fixed-width binary index.

    The complete index is reconstructed in memory.

    Args:
        directory: Directory containing a saved binary index.

    Returns:
        An InvertedIndex with tuple-based posting lists.

    Raises:
        FileNotFoundError: If a required file is missing.
        ValueError: If metadata, tokenizer settings, binary layout,
            or index contents are inconsistent.
    """
    directory = Path(directory)

    required_files = [
        "manifest.json",
        "tokenizer_config.json",
        "vocabulary.json",
        "doc_ids.json",
        "doc_lengths.bin",
        "postings.bin",
    ]

    for name in required_files:
        path = directory / name

        if not path.is_file():
            raise FileNotFoundError(
                f"Required index file is missing: {path}"
            )

    manifest = _read_json(directory / "manifest.json")

    if not isinstance(manifest, dict):
        raise ValueError("Index manifest must be a JSON object.")

    if manifest.get("format") != FORMAT_NAME:
        raise ValueError("Unsupported binary index format.")

    if manifest.get("format_version") != FORMAT_VERSION:
        raise ValueError("Unsupported binary index format version.")

    verify_tokenizer_config(
        directory / "tokenizer_config.json"
    )

    document_ids = _read_json(directory / "doc_ids.json")
    vocabulary = _read_json(directory / "vocabulary.json")

    if not isinstance(document_ids, list):
        raise ValueError("Document IDs must be a JSON list.")

    if not isinstance(vocabulary, dict):
        raise ValueError("Vocabulary must be a JSON object.")

    document_count = len(document_ids)

    if manifest.get("document_count") != document_count:
        raise ValueError(
            "Document count does not match the manifest."
        )

    if manifest.get("term_count") != len(vocabulary):
        raise ValueError(
            "Term count does not match the manifest."
        )

    lengths_path = directory / "doc_lengths.bin"
    expected_length_bytes = document_count * UINT32.size

    if lengths_path.stat().st_size != expected_length_bytes:
        raise ValueError("Invalid document-length file size.")

    lengths_data = lengths_path.read_bytes()

    document_lengths = [
        length
        for (length,) in UINT32.iter_unpack(lengths_data)
    ]

    postings_path = directory / "postings.bin"
    postings_file_size = postings_path.stat().st_size

    postings = {}

    with postings_path.open("rb") as file:
        for term in sorted(vocabulary):
            entry = vocabulary[term]

            if not isinstance(entry, dict):
                raise ValueError(
                    f"Invalid vocabulary entry for {term!r}."
                )

            offset = entry.get("offset")
            document_frequency = entry.get(
                "document_frequency"
            )

            if (
                type(offset) is not int
                or offset != file.tell()
            ):
                raise ValueError(
                    f"Invalid posting offset for {term!r}."
                )

            if (
                type(document_frequency) is not int
                or not 1 <= document_frequency <= document_count
            ):
                raise ValueError(
                    f"Invalid document frequency for {term!r}."
                )

            byte_count = document_frequency * POSTING.size

            if offset + byte_count > postings_file_size:
                raise ValueError(
                    f"Truncated postings for {term!r}."
                )

            block = file.read(byte_count)

            if len(block) != byte_count:
                raise ValueError(
                    f"Truncated postings for {term!r}."
                )

            postings[term] = list(
                POSTING.iter_unpack(block)
            )

        if file.tell() != postings_file_size:
            raise ValueError(
                "Unexpected trailing data in postings.bin."
            )

    index = InvertedIndex(
        postings=postings,
        doc_lengths=document_lengths,
        internal_to_external=document_ids,
    )

    validate_index(index)

    return index

    class BM25:
    """Score matching documents with the Okapi BM25 formula.

    Args:
        index: In-memory index whose postings and document statistics are
            treated as read-only for this ranker's lifetime.
        k1: Positive finite term-frequency saturation parameter (default 1.2).
        b: Finite document-length normalization parameter in ``[0, 1]``
            (default 0.75).

    The implementation uses ``log(1 + (N - df + 0.5) / (df + 0.5))`` for
    IDF and sums the standard BM25 term contribution. Repeated normalized
    query terms count once. Results sort by descending score, with ties broken
    by ascending internal ID, preserving input order independently of external
    ID spelling. Zero-score documents are omitted.
    """

    def __init__(
        self,
        index: InvertedIndex,
        k1: float = 1.2,
        b: float = 0.75,
    ):
        """Create a ranker; invalid ``k1`` or ``b`` raises ``ValueError``."""
        if not math.isfinite(k1) or k1 <= 0:
            raise ValueError("k1 must be finite and greater than zero.")

        if not math.isfinite(b) or not 0 <= b <= 1:
            raise ValueError("b must be finite and between 0 and 1.")

        self.index = index
        self.k1 = k1
        self.b = b

        # Compute these once, rather than repeatedly during scoring.
        self.document_count = index.document_count
        self.average_document_length = index.average_document_length

    def search(
        self,
        query: str,
        top_k: int = 10,
    ) -> list[SearchResult]:
        """Return up to ``top_k`` positive-scoring matches for ``query``.

        Args:
            query: Text normalized with the index tokenizer.
            top_k: Maximum result count; nonpositive values return no results.

        Returns:
            Results containing internal ID, original external ``doc_id``, and
            BM25 score, in descending score order with deterministic ID ties.
            Empty/unknown queries and empty or all-zero-length collections
            return an empty list. Query terms are deduplicated after
            tokenization.

        Raises:
            TypeError: If ``query`` is not a string.
        """
        if top_k <= 0:
            return []

        # Sorting gives a deterministic accumulation order.
        query_terms = sorted(set(tokenize(query)))

        if not query_terms:
            return []

        if (
            self.document_count == 0
            or self.average_document_length == 0
        ):
            return []

        scores: dict[int, float] = defaultdict(float)

        for term in query_terms:
            postings = self.index.postings.get(term, [])

            if not postings:
                continue

            document_frequency = len(postings)

            idf = math.log1p(
                (
                    self.document_count
                    - document_frequency
                    + 0.5
                )
                / (document_frequency + 0.5)
            )

            for internal_id, term_frequency in postings:
                document_length = self.index.doc_lengths[internal_id]

                length_normalization = (
                    1
                    - self.b
                    + self.b
                    * document_length
                    / self.average_document_length
                )

                numerator = term_frequency * (self.k1 + 1)

                denominator = (
                    term_frequency
                    + self.k1 * length_normalization
                )

                scores[internal_id] += (
                    idf * numerator / denominator
                )

        # Smallest (-score, ID) means highest score, then lowest ID.
        best = heapq.nsmallest(
            top_k,
            scores.items(),
            key=lambda item: (-item[1], item[0]),
        )

        return [
            SearchResult(
                internal_id=internal_id,
                doc_id=self.index.internal_to_external[internal_id],
                score=score,
            )
            for internal_id, score in best
        ]

    def read_jsonl(path):
    """Read nonblank JSON Lines records with their source line numbers.

    Args:
        path (Path): UTF-8 JSONL file to read.

    Each yielded pair contains the source line number and decoded JSON object.

    Raises:
        ValueError: If a nonblank line is invalid JSON or is not a JSON object.
    """
    with path.open("r", encoding="utf-8") as file:
        for line_number, line in enumerate(file, start=1):
            if not line.strip():
                continue

            try:
                row = json.loads(line)
            except json.JSONDecodeError as exc:
                raise ValueError(
                    f"Invalid JSON in {path}, line {line_number}"
                ) from exc

            if not isinstance(row, dict):
                raise ValueError(
                    f"Expected a JSON object in {path}, "
                    f"line {line_number}"
                )

            yield line_number, row

    """Convert the raw SciFact corpus and create its external ID mapping.

    Returns the set of document IDs and the number of empty documents.

    Raises:
        ValueError: If a record is invalid or contains a duplicate document ID.
    """
    input_path = RAW_DIR / "corpus.jsonl"
    documents_path = OUTPUT_DIR / "documents.jsonl"
    mapping_path = OUTPUT_DIR / "doc_ids.jsonl"

    seen_ids = set()
    empty_documents = 0

    with (
        documents_path.open("w", encoding="utf-8") as documents_file,
        mapping_path.open("w", encoding="utf-8") as mapping_file,
    ):
        for line_number, row in read_jsonl(input_path):
            location = f"{input_path}:{line_number}"

            doc_id = require_id(row, "_id", location)
            title = require_text(row, "title", location, default="")
            text = require_text(row, "text", location)

                raise ValueError(
                    f"{location}: duplicate document ID '{doc_id}'"
                )

            internal_id = len(seen_ids)
            seen_ids.add(doc_id)

            if not title.strip() and not text.strip():
                empty_documents += 1

    """Convert raw SciFact queries into the processed JSONL format.

                "text": text,

    def convert_qrels(document_ids, query_ids):
    """Convert qrels splits and validate referenced queries and documents.

    Args:
        document_ids (set[str]): Valid document IDs from the converted corpus.
        query_ids (set[str]): Valid query IDs from the converted queries.

    Returns per-split counts of judgments, queries, and positive judgments.
    Writes one normalized TSV under ``data/processed/scifact/qrels`` for each
    raw split.

    Raises:
        ValueError: If qrels are missing, malformed, duplicated, or reference unknown IDs.
    """
    input_dir = RAW_DIR / "qrels"
    output_dir = OUTPUT_DIR / "qrels"

    output_dir.mkdir(parents=True, exist_ok=True)

    split_paths = sorted(input_dir.glob("*.tsv"))

    if not split_paths:
        raise ValueError(f"No qrels files found in {input_dir}")

    split_statistics = {}

    for input_path in split_paths:
        split_name = input_path.stem
        output_path = output_dir / input_path.name

        seen_pairs = set()
        split_query_ids = set()
        positive_count = 0

        with (
            input_path.open(
                "r", encoding="utf-8", newline=""
            ) as input_file,
            output_path.open(
                "w", encoding="utf-8", newline=""
            ) as output_file,
        ):
            reader = csv.DictReader(input_file, delimiter="\t")

            expected_fields = {"query-id", "corpus-id", "score"}
            actual_fields = set(reader.fieldnames or [])

            if not expected_fields.issubset(actual_fields):
                raise ValueError(
                    f"{input_path}: unexpected columns "
                    f"{reader.fieldnames}"
                )

            writer = csv.DictWriter(
                output_file,
                fieldnames=["query_id", "doc_id", "relevance"],
                delimiter="\t",
                lineterminator="\n",
            )
            writer.writeheader()

            for row in reader:
                location = f"{input_path}:{reader.line_num}"

                query_id = row["query-id"]
                doc_id = row["corpus-id"]

                try:
                    relevance = int(row["score"])
                except (ValueError, TypeError) as exc:
                    raise ValueError(
                        f"{location}: invalid relevance score"
                    ) from exc

                if query_id not in query_ids:
                    raise ValueError(
                        f"{location}: unknown query '{query_id}'"
                    )

                if doc_id not in document_ids:
                    raise ValueError(
                        f"{location}: unknown document '{doc_id}'"
                    )

                pair = (query_id, doc_id)

                if pair in seen_pairs:
                    raise ValueError(
                        f"{location}: duplicate judgment {pair}"
                    )

                seen_pairs.add(pair)
                split_query_ids.add(query_id)

                if relevance > 0:
                    positive_count += 1

                writer.writerow({
                    "query_id": query_id,
                    "doc_id": doc_id,
                    "relevance": relevance,
                })

        split_statistics[split_name] = {
            "judgments": len(seen_pairs),
            "queries": len(split_query_ids),
            "positive_judgments": positive_count,
        }

    return split_statistics

    """Convert raw SciFact files and write processed data with a manifest.

    Raises:
        FileNotFoundError: If a required raw input path does not exist.
        FileExistsError: If the output directory already exists.
        ValueError: If converted collections are empty or contain invalid records.

    Side effects:
        Creates a new ``data/processed/scifact`` tree containing converted
        documents, queries, ID mappings, qrels, and a manifest. It refuses to
        overwrite an existing output directory, and prints a conversion
        summary when complete.

    required_paths = [
        RAW_DIR / "corpus.jsonl",
        RAW_DIR / "queries.jsonl",
        RAW_DIR / "qrels",
    ]

    for path in required_paths:
        if not path.exists():
            raise FileNotFoundError(
                f"Required input not found: {path}"
            )

    if OUTPUT_DIR.exists():
        raise FileExistsError(
            f"Output directory already exists: {OUTPUT_DIR}\n"
            "Inspect it before removing or renaming it."
        )

    OUTPUT_DIR.mkdir(parents=True)

    document_ids, empty_documents = convert_documents()
    query_ids = convert_queries()

    if not document_ids:
        raise ValueError("The document collection is empty.")

    if not query_ids:
        raise ValueError("The query collection is empty.")

    split_statistics = convert_qrels(
        document_ids,
        query_ids,
    )

    manifest = {
        "dataset": "scifact",
        "schema_version": SCHEMA_VERSION,
        "created_at_utc": datetime.now(timezone.utc).isoformat(),
        "input_directory": str(RAW_DIR.relative_to(PROJECT_ROOT)),
        "documents": len(document_ids),
        "queries": len(query_ids),
        "empty_documents": empty_documents,
        "internal_id_assignment": "zero-based corpus file order",
        "splits": split_statistics,
    }

    manifest_path = OUTPUT_DIR / "manifest.json"

    with manifest_path.open("w", encoding="utf-8") as file:
        json.dump(
            manifest,
            file,
            indent=2,
            ensure_ascii=False,
        )

    print(f"Documents: {len(document_ids):,}")
    print(f"Queries: {len(query_ids):,}")
    print(f"Empty documents: {empty_documents:,}")

    for split_name, stats in split_statistics.items():
        print(
            f"{split_name}: "
            f"{stats['queries']:,} queries, "
            f"{stats['judgments']:,} judgments"
        )

    def main():
    """Build an in-memory index and validate it against the processed manifest.

    Reads ``documents.jsonl``, ``doc_ids.jsonl``, and ``manifest.json``;
    prints index statistics and sample postings. Does not persist the index.

    Raises:
        FileNotFoundError: If any required processed file is missing.
        ValueError: If counts or internal/external ID mappings disagree.
    """
    documents_path = DATA_DIR / "documents.jsonl"
    mapping_path = DATA_DIR / "doc_ids.jsonl"
    manifest_path = DATA_DIR / "manifest.json"

    for path in (documents_path, mapping_path, manifest_path):
        if not path.is_file():
            raise FileNotFoundError(
                f"Missing {path}. Complete Step 4 first."
            )

    # build_index accepts an iterable, so it can consume the
    # documents produced by read_jsonl one at a time.
    index = build_index(read_jsonl(documents_path))

    with manifest_path.open("r", encoding="utf-8") as file:
        manifest = json.load(file)

    expected_count = manifest["documents"]

    if index.document_count != expected_count:
        raise ValueError(
            f"Document count mismatch: index has "
            f"{index.document_count}, manifest says {expected_count}"
        )

    # Step 4 assigned internal IDs in corpus-file order.
    # Confirm that build_index assigned exactly the same IDs.
    mapping_count = 0

    for mapping in read_jsonl(mapping_path):
        expected_internal_id = mapping_count
        actual_internal_id = mapping["internal_id"]

        if actual_internal_id != expected_internal_id:
            raise ValueError(
                f"Mapping entry {mapping_count} has internal ID "
                f"{actual_internal_id}, expected {expected_internal_id}"
            )

        if expected_internal_id >= index.document_count:
            raise ValueError(
                "ID mapping contains more documents than the index"
            )

        expected_doc_id = mapping["doc_id"]
        actual_doc_id = index.internal_to_external[
            expected_internal_id
        ]

        if actual_doc_id != expected_doc_id:
            raise ValueError(
                f"ID mismatch at position {expected_internal_id}: "
                f"index has {actual_doc_id!r}, "
                f"mapping has {expected_doc_id!r}"
            )

        mapping_count += 1

    if mapping_count != index.document_count:
        raise ValueError(
            f"ID mapping has {mapping_count} documents; "
            f"index has {index.document_count}"
        )

    print("SciFact index checks passed.")
    print(f"Documents: {index.document_count:,}")
    print(f"Distinct terms: {len(index.postings):,}")
    print(
        "Average document length: "
        f"{index.average_document_length:.2f} tokens"
    )

    print("\nExample terms:")
    for term in ("science", "evidence", "protein"):
        postings = index.postings.get(term, [])
        print(
            f"  {term!r}: {len(postings)} documents; "
            f"first five postings: {postings[:5]}"
        )

    def main():
    """Load a saved index, run BM25, and display ranked SciFact results."""
    parser = argparse.ArgumentParser(
        description="Search SciFact using stratajanus BM25."
    )

    parser.add_argument(
        "query",
        help="Search query. Put multiword queries in quotes.",
    )

    parser.add_argument(
        "--top-k",
        type=int,
        default=10,
        help="Maximum number of results.",
    )

    parser.add_argument(
        "--format",
        choices=["json", "fixed32", "vb"],
        default="json",
        help="Saved index format to load.",
    )

    args = parser.parse_args()

    if args.format == "vb":
        index_directory = COMPRESSED_INDEX_DIR
        loader = load_compressed_index
        build_command = "python scripts/compress_scifact_index.py"
    elif args.format == "fixed32":
        index_directory = BINARY_INDEX_DIR
        loader = load_binary_index
        build_command = "python scripts/convert_index_to_binary.py"
    else:
        index_directory = JSON_INDEX_DIR
        loader = load_json_index
        build_command = "python scripts/build_scifact_index.py"

    if not index_directory.is_dir():
        raise FileNotFoundError(
            f"No saved index found at: {index_directory}\n"
            f"Run: {build_command}"
        )

    print(f"Loading {args.format} index...")
    index = loader(index_directory)

    ranker = BM25(index)

    results = ranker.search(
        args.query,
        top_k=args.top_k,
    )

    titles = find_titles(
        {result.doc_id for result in results}
    )

    print(f"\nQuery: {args.query}")
    print(f"Results returned: {len(results)}\n")

    for rank, result in enumerate(results, start=1):
        print(
            f"{rank}. {result.doc_id} "
            f"| BM25 score: {result.score:.4f}"
        )

        title = titles.get(result.doc_id)

        if title:
            print(f"   {title}")

        print()
